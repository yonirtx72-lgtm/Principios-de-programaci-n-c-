01 // VARIABLES

Variables

Una variable es un espacio de memoria que almacena un dato que puede cambiar durante la ejecución. 

int edad = 15;

cout << edad;
02 // TIPOS DE DATOS

Tipos de datos

Los tipos de datos indican qué clase de información puede almacenar una variable. 

int edad = 15;
double precio = 25.5;
char letra = 'A';
bool activo = true;
03 // OPERADORES

Operadores

Permiten realizar operaciones matemáticas y lógicas. 

int a = 10;
int b = 5;

cout << a + b;
cout << a - b;
cout << a * b;
cout << a / b;
04 // ENTRADA Y SALIDA

Entrada y salida

C++ utiliza cin para recibir datos y cout para mostrar información en pantalla. 

int edad;

cout << "Edad: ";
cin >> edad;

cout << "Tienes "
     << edad
     << " años";
05 // CONDICIONALES

if / else

Permiten ejecutar diferentes instrucciones dependiendo de una condición. 

int edad = 18;

if(edad >= 18){

    cout << "Mayor de edad";

}else{

    cout << "Menor de edad";

}
06 // SWITCH

Switch

Switch permite seleccionar una opción entre varios casos. 

int opcion = 2;

switch(opcion){

case 1:
    cout << "Inicio";
    break;

case 2:
    cout << "Configuración";
    break;

default:
    cout << "Opción inválida";

}
07 // FOR

Bucle for

Permite repetir instrucciones una cantidad determinada de veces. 

for(int i = 1; i <= 5; i++){

    cout << i << endl;

}
08 // WHILE

While

Repite instrucciones mientras una condición sea verdadera. 

int numero = 1;

while(numero <= 5){

    cout << numero << endl;

    numero++;

}
09 // DO WHILE

Do while

Ejecuta primero el código y después comprueba la condición. 

int numero = 1;

do{

    cout << numero;
    numero++;

}while(numero <= 5);
10 // FUNCIONES

Funciones

Una función agrupa instrucciones que pueden reutilizarse. 

int sumar(int a, int b){

    return a + b;

}

cout << sumar(5,10);
11 // REFERENCIAS

Referencias

Una referencia es otro nombre para una variable existente. 

int numero = 10;

int &referencia = numero;

referencia = 20;

cout << numero;
12 // PUNTEROS

Punteros

Un puntero almacena la dirección de memoria de otra variable. 

int numero = 10;

int *puntero = &numero;

cout << *puntero;
13 // CLASES

Clases y objetos

Las clases permiten crear estructuras que combinan datos y funciones. 

class Persona{

public:

    string nombre;

    void saludar(){

        cout << "Hola";

    }

};

Persona persona;

persona.saludar();
14 // HERENCIA

Herencia

Permite que una clase reutilice características de otra. 

class Animal{

public:

    void comer(){

        cout << "Comiendo";

    }

};

class Perro : public Animal{

public:

    void ladrar(){

        cout << "Guau";

    }

};
15 // POLIMORFISMO

Polimorfismo

Permite que diferentes clases respondan de diferentes maneras a una misma función. 

class Animal{

public:

    virtual void sonido(){

        cout << "Sonido";

    }

};

class Perro : public Animal{

public:

    void sonido() override{

        cout << "Guau";

    }

};
16 // STL

Standard Template Library

La STL proporciona contenedores, algoritmos y herramientas para trabajar de forma más eficiente. 

#include <vector>
#include <algorithm>

vector<int> numeros = {
    5,2,8,1
};

sort(
    numeros.begin(),
    numeros.end()
);
17 // TEMPLATES

Templates

Permiten crear funciones y clases que trabajan con diferentes tipos de datos. 

template <typename T>

T mayor(T a, T b){

    return (a > b) ? a : b;

}

int main(){

    cout << mayor(10, 20);

    return 0;

}
18 // LAMBDAS

Funciones Lambda

Las funciones lambda permiten crear pequeñas funciones directamente dentro del código. 

auto sumar = [](int a, int b){

    return a + b;

};

cout << sumar(5, 10);
19 // SMART POINTERS

Smart Pointers

Los punteros inteligentes ayudan a administrar automáticamente la memoria. 

#include <memory>

auto numero =
std::make_unique<int>(100);

cout << *numero;
20 // THREADS

Threads

Los threads permiten ejecutar diferentes tareas de manera concurrente. 

#include <thread>
#include <iostream>

void tarea(){

    std::cout << "Ejecutando...";

}

int main(){

    std::thread hilo(tarea);

    hilo.join();

    return 0;

}
21 // STRUCT

Struct

Una estructura permite agrupar diferentes tipos de datos bajo un mismo nombre. 

struct Jugador{

    string nombre;

    int nivel;

    int puntos;

};

Jugador jugador;

jugador.nombre = "Yoni";
jugador.nivel = 10;
jugador.puntos = 500;
22 // ENUM

Enumeraciones

Un enum permite crear un conjunto de valores constantes con nombres fáciles de entender. 

enum Estado{

    INICIO,

    JUGANDO,

    PAUSA,

    FINALIZADO

};

Estado estado = JUGANDO;
23 // CONST

Constantes

La palabra const permite declarar valores que no deben modificarse durante el programa. 

const double PI = 3.14159;

const int MAX_JUGADORES = 10;

cout << PI;
24 // STATIC

Static

Static permite conservar el valor de una variable entre diferentes llamadas a una función. 

void contador(){

    static int numero = 0;

    numero++;

    cout << numero;

}

int main(){

    contador();

    contador();

    contador();

}
25 // NAMESPACE

Namespaces

Los namespaces permiten organizar código y evitar conflictos entre nombres. 

namespace Juego{

    int puntos = 100;

}

namespace Jugador{

    int puntos = 500;

}

cout << Juego::puntos;

cout << Jugador::puntos;
26 // USING

Using

Using permite utilizar nombres de namespaces o crear alias de tipos de forma más sencilla. 

#include <iostream>

using std::cout;

int main(){

    cout << "Hola C++";

    return 0;

}
27 // CONSTRUCTORES

Constructores

Un constructor se ejecuta automáticamente cuando se crea un objeto. 

class Persona{

public:

    Persona(){

        cout << "Objeto creado";

    }

};

int main(){

    Persona persona;

}
28 // DESTRUCTORES

Destructores

Un destructor se ejecuta automáticamente cuando un objeto es destruido. 

class Persona{

public:

    ~Persona(){

        cout << "Objeto destruido";

    }

};

int main(){

    Persona persona;

}
29 // ENCAPSULAMIENTO

Encapsulamiento

El encapsulamiento permite proteger los datos internos de una clase mediante modificadores como private y public. 

class Cuenta{

private:

    double saldo = 0;

public:

    void depositar(double cantidad){

        saldo += cantidad;

    }

    double obtenerSaldo(){

        return saldo;

    }

};
30 // ABSTRACCIÓN

Abstracción

La abstracción permite mostrar solamente lo necesario y ocultar detalles internos de implementación. 

class Vehiculo{

public:

    void arrancar(){

        encenderMotor();

        cout << "Vehiculo listo";

    }

private:

    void encenderMotor(){

        cout << "Motor encendido";

    }

};
31 // SOBRECARGA

Sobrecarga de funciones

Permite utilizar el mismo nombre de función con diferentes tipos o cantidades de parámetros. 

int sumar(int a, int b){

    return a + b;

}

double sumar(double a, double b){

    return a + b;

}

cout << sumar(10,20);

cout << sumar(2.5,3.5);
32 // SOBRECARGA DE OPERADORES

Sobrecarga de operadores

Permite definir cómo funcionan operadores como +, -, == o < con objetos personalizados. 

class Numero{

public:

    int valor;

    Numero operator+(Numero otro){

        Numero resultado;

        resultado.valor =
            valor + otro.valor;

        return resultado;

    }

};
33 // THIS

this

this es un puntero que hace referencia al objeto actual. 

class Persona{

private:

    int edad;

public:

    void setEdad(int edad){

        this->edad = edad;

    }

};
34 // CLASES ABSTRACTAS

Clases abstractas

Una clase abstracta sirve como base para otras clases y puede contener funciones virtuales puras. 

class Animal{

public:

    virtual void sonido() = 0;

};

class Perro : public Animal{

public:

    void sonido() override{

        cout << "Guau";

    }

};
35 // VIRTUAL

Funciones virtuales

Las funciones virtuales permiten aplicar polimorfismo durante la ejecución del programa. 

class Animal{

public:

    virtual void sonido(){

        cout << "Sonido";

    }

};

class Gato : public Animal{

public:

    void sonido() override{

        cout << "Miau";

    }

};
36 // OVERRIDE

Override

Override indica explícitamente que una función sobrescribe una función virtual de la clase base. 

class Animal{

public:

    virtual void sonido(){

        cout << "Sonido";

    }

};

class Perro : public Animal{

public:

    void sonido() override{

        cout << "Guau";

    }

};
37 // FRIEND

Friend

Friend permite que una función o clase externa tenga acceso a miembros privados. 

class Persona{

private:

    int edad = 15;

    friend void mostrar(Persona p);

};

void mostrar(Persona p){

    cout << p.edad;

}
38 // EXCEPCIONES

Excepciones

Las excepciones permiten manejar situaciones inesperadas durante la ejecución de un programa. 

try{

    throw "Ocurrio un error";

}catch(const char* mensaje){

    cout << mensaje;

}
39 // TRY CATCH

Try / Catch

Try contiene código que puede producir una excepción y catch permite manejarla. 

try{

    int edad = -5;

    if(edad < 0){

        throw edad;

    }

}catch(int error){

    cout <<
        "Edad invalida";

}
40 // ARCHIVOS

Archivos

C++ puede guardar información permanentemente utilizando archivos. 

#include <fstream>

ofstream archivo("datos.txt");

archivo << "Hola desde C++";

archivo.close();
41 // FSTREAM

fstream

fstream permite abrir archivos para realizar operaciones de lectura y escritura. 

#include <fstream>

fstream archivo;

archivo.open(
    "datos.txt",
    ios::in | ios::out
);

archivo.close();
42 // STRING

String

string permite almacenar y manipular cadenas de texto. 

#include <string>

string nombre = "Yoni";

cout << nombre;

cout << nombre.length();
43 // VECTOR

Vector

vector es un contenedor dinámico que puede almacenar varios elementos del mismo tipo. 

#include <vector>

vector<int> numeros;

numeros.push_back(10);
numeros.push_back(20);
numeros.push_back(30);

cout << numeros[0];
44 // MAP

Map

map almacena pares formados por una clave y un valor. 

#include <map>

map<string,int> niveles;

niveles["Yoni"] = 50;

niveles["Alex"] = 30;

cout <<
    niveles["Yoni"];
45 // SET

Set

set almacena elementos únicos y automáticamente evita duplicados. 

#include <set>

set<int> numeros;

numeros.insert(10);
numeros.insert(10);
numeros.insert(20);

cout <<
    numeros.size();
46 // STACK

Stack

Stack utiliza el principio LIFO: el último elemento que entra es el primero que sale. 

#include <stack>

stack<int> pila;

pila.push(10);
pila.push(20);
pila.push(30);

cout << pila.top();

pila.pop();
47 // QUEUE

Queue

Queue utiliza el principio FIFO: el primer elemento que entra es el primero que sale. 

#include <queue>

queue<int> cola;

cola.push(10);
cola.push(20);
cola.push(30);

cout << cola.front();

cola.pop();
48 // ALGORITMOS STL

Algoritmos de la STL

La STL incluye algoritmos para ordenar, buscar y modificar colecciones de datos. 

#include <algorithm>
#include <vector>

vector<int> numeros =
{5,2,8,1,4};

sort(
    numeros.begin(),
    numeros.end()
);

for(int n : numeros){

    cout << n << " ";

}
49 // ITERADORES

Iteradores

Los iteradores permiten recorrer y manipular elementos de los contenedores de la STL. 

vector<int> numeros =
{10,20,30};

for(
    auto it = numeros.begin();
    it != numeros.end();
    ++it
){

    cout << *it << " ";

}
50 // BUENAS PRÁCTICAS

Buenas prácticas

Las buenas prácticas ayudan a crear programas más ordenados, legibles, fáciles de mantener y con menos errores. 

#include <iostream>

using namespace std;

int sumar(int a, int b){

    return a + b;

}

int main(){

    int resultado =
        sumar(10, 20);

    cout << resultado;

    return 0;

}
